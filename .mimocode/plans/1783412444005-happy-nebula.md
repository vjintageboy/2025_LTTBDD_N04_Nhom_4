# Prisma Schema Management Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use compose:subagent (recommended) or compose:execute to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add Prisma for schema management only - document and manage the PostgreSQL schema through schema.prisma while keeping Supabase for queries.

**Architecture:** 
- Create a `prisma/` directory at project root with schema.prisma
- Schema mirrors current 13 tables in Supabase PostgreSQL
- Prisma CLI used for schema validation, documentation, and potential migration generation
- Supabase client remains the primary data access layer in Flutter

**Tech Stack:** Prisma ORM (schema only), PostgreSQL (Supabase), Flutter + Supabase client

## Global Constraints

- Prisma is for schema management ONLY - no backend API changes
- Supabase client remains the data access layer
- All 13 current tables must be represented in schema.prisma
- Custom PostgreSQL types (user_role, appointment_status, call_type) must be defined
- Vector columns (embedding) from migrations should be documented
- Keep schema.prisma in sync with supabase_schema.sql

```prisma

}

  provider = "postgresql"
  url      = env("DATABASE_URL")
}
```

model User {
  id            String    @id @default(uuid()) @db.Uuid
  email         String    @unique @db.VarChar(255)
  fullName      String    @map("full_name") @db.VarChar(255)
  avatarUrl     String?   @map("avatar_url") @db.Text
  role          UserRole  @default(user)
  streakCount   Int       @map("streak_count") @default(0)
  longestStreak Int       @map("longest_streak") @default(0)
  totalActivities Int     @map("total_activities") @default(0)
  lastLogin     DateTime? @map("last_login") @db.Timestamptz
  dateOfBirth   DateTime? @map("date_of_birth") @db.Timestamptz
  gender        String?   @db.VarChar(50)
  goals         String[]  @db.Array(String)
  preferences   Json?
  createdAt     DateTime  @default(now()) @map("created_at") @db.Timestamptz
  updatedAt     DateTime  @updatedAt @map("updated_at") @db.Timestamptz

  // Relations
  moodEntries     MoodEntry[]
  appointments    Appointment[]
  chatParticipants ChatParticipant[]
  posts           Post[]
  postLikes       PostLike[]
  postComments    PostComment[]
  notifications   Notification[]
  messages        Message[]

  @@map("users")
}

model MoodEntry {
  id             String   @id @default(uuid()) @db.Uuid
  userId         String   @map("user_id") @db.Uuid
  moodScore      Int      @map("mood_score") @db.Check("mood_score >= 1 AND mood_score <= 5")
  note           String?  @db.Text
  emotionFactors String[] @map("emotion_factors") @db.Array(String)
  tags           String[] @db.Array(String)
  createdAt      DateTime @default(now()) @map("created_at") @db.Timestamptz
  updatedAt      DateTime @updatedAt @map("updated_at") @db.Timestamptz

  // Relations
  user User @relation(fields: [userId], references: [id])

  @@index([userId])
  @@map("mood_entries")
}

model Meditation {
  id             String   @id @default(uuid()) @db.Uuid
  title          String   @db.VarChar(255)
  description    String?  @db.Text
  category       String?  @db.VarChar(100)
  durationMinutes Int    @map("duration_minutes")
  audioUrl       String   @map("audio_url") @db.Text
  thumbnailUrl   String?  @map("thumbnail_url") @db.Text
  level          String   @default("beginner") @db.VarChar(50)
  rating         Decimal  @default(0) @db.Numeric(3, 2)
  totalReviews   Int      @map("total_reviews") @default(0)
  createdAt      DateTime @default(now()) @map("created_at") @db.Timestamptz
  updatedAt      DateTime @updatedAt @map("updated_at") @db.Timestamptz

  @@map("meditations")
}

model ChatParticipant {
  roomId          String    @map("room_id") @db.Uuid

  lastReadAt      DateTime? @map("last_read_at") @db.Timestamptz

model Message {

  createdAt         DateTime  @default(now()) @map("created_at") @db.Timestamptz
  updatedAt         DateTime  @updatedAt @map("updated_at") @db.Timestamptz

  // Relations

  sender User     @relation(fields: [senderId], references: [id])

model Post {
  id           String   @id @default(uuid()) @db.Uuid
  authorId     String?  @map("author_id") @db.Uuid
  title        String   @db.VarChar(255)
  content      String   @db.Text
  imageUrl     String?  @map("image_url") @db.Text
  likesCount   Int      @map("likes_count") @default(0)
  commentCount Int      @map("comment_count") @default(0)
  category     String   @default("community") @db.VarChar(100)
  isAnonymous  Boolean  @map("is_anonymous") @default(false)
  createdAt    DateTime @default(now()) @map("created_at") @db.Timestamptz
  updatedAt    DateTime @updatedAt @map("updated_at") @db.Timestamptz

  // Relations
  author   User?          @relation(fields: [authorId], references: [id])
  likes    PostLike[]
  comments PostComment[]

  @@index([authorId])
  @@map("posts")
}

model PostLike {
  id        String   @id @default(uuid()) @db.Uuid
  postId    String   @map("post_id") @db.Uuid
  userId    String   @map("user_id") @db.Uuid
  createdAt DateTime @default(now()) @map("created_at") @db.Timestamptz

  // Relations
  post Post @relation(fields: [postId], references: [id])
  user User @relation(fields: [userId], references: [id])

  @@unique([postId, userId])
  @@map("post_likes")
}

model PostComment {
  id              String   @id @default(uuid()) @db.Uuid
  postId          String   @map("post_id") @db.Uuid
  userId          String   @map("user_id") @db.Uuid
  content         String   @db.Text
  isAnonymous     Boolean  @map("is_anonymous") @default(false)
  parentCommentId String?  @map("parent_comment_id") @db.Uuid
  createdAt       DateTime @default(now()) @map("created_at") @db.Timestamptz
  updatedAt       DateTime @updatedAt @map("updated_at") @db.Timestamptz

  // Relations
  post            Post          @relation(fields: [postId], references: [id])
  user            User          @relation(fields: [userId], references: [id])
  parentComment   PostComment?  @relation("CommentReplies", fields: [parentCommentId], references: [id])
  replies         PostComment[] @relation("CommentReplies")

  @@index([postId])
  @@index([parentCommentId])
  @@map("post_comments")
}

model Notification {
  id        String   @id @default(uuid()) @db.Uuid
  userId    String   @map("user_id") @db.Uuid
  title     String   @db.VarChar(255)
  message   String   @db.Text
  type      String   @db.VarChar(50)
  isRead    Boolean  @map("is_read") @default(false)
  createdAt DateTime @default(now()) @map("created_at") @db.Timestamptz

  // Relations
  user User @relation(fields: [userId], references: [id])

  @@index([userId])
  @@map("notifications")

model Expert {

  yearsExperience  Int       @map("years_experience") @default(0)
  licenseNumber    String?   @map("license_number") @db.VarChar(255)

model ExpertAvailability {

  refundStatus    String?           @map("refund_status") @default("none") @db.VarChar(50)
  cancelledRole   String?           @map("cancelled_role") @db.VarChar(50)

  updatedAt       DateTime          @updatedAt @map("updated_at") @db.Timestamptz
